cmake_minimum_required(VERSION 3.20)
project(vibration_verification LANGUAGES CXX)
include(CTest)
function(verified_target name source)
  add_executable(${name} ${source})
  target_compile_features(${name} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${name} PRIVATE /W4 /WX)
  else()
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${name} COMMAND ${name})
endfunction()
verified_target(streaming_invariants modules/aliasguard/cpp/test_fir.cpp)
verified_target(native_state modules/forcewindow/cpp/test_state.cpp)
verified_target(native_energy modules/jointguard/cpp/test_energy.cpp)
verified_target(oscillator_handoff cpp/oscillator_handoff.cpp)
