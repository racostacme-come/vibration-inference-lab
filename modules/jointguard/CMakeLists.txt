cmake_minimum_required(VERSION 3.20)
project(jointguard LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_core cpp/bindings.cpp)
target_compile_features(_core PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(_core PRIVATE /W4 /WX)
else()
  target_compile_options(_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
install(TARGETS _core DESTINATION jointguard)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_energy cpp/test_energy.cpp)
  target_compile_features(test_energy PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(test_energy PRIVATE /W4 /WX)
  else()
    target_compile_options(test_energy PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME native_energy COMMAND test_energy)
endif()
